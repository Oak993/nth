// VT-0 evidence probe (Q-V-9 C feasibility): does Whisper.net 1.9.1 load a runtime from a folder it is pointed at, and what does it do when
// a CUDA runtime cannot load? A scratch console, never part of the app and never run by a test. Usage:
//   probe <libraryPath|-> <order, comma separated RuntimeLibrary names|-> <gpu|cpu> <vad model path> <16 kHz mono 16-bit wav>
// It prints the options it was given, the runtime the loader reports (RuntimeOptions.LoadedLibrary) after the Silero VAD model has been
// loaded through WhisperVadFactory, and the speech segments the VAD finds in the clip, or the failure.
using Whisper.net;
using Whisper.net.LibraryLoader;

var libPath = args[0] != "-" ? args[0] : null;
var order = args[1] != "-" ? args[1].Split(',').Select(Enum.Parse<RuntimeLibrary>).ToList() : null;
var useGpu = args[2] == "gpu";
if (libPath is not null) RuntimeOptions.LibraryPath = libPath;
if (order is not null) RuntimeOptions.RuntimeLibraryOrder = order;
Console.WriteLine($"LibraryPath={RuntimeOptions.LibraryPath ?? "(null)"} order={string.Join(",", RuntimeOptions.RuntimeLibraryOrder ?? new())} useGpu={useGpu}");
try
{
    using var vad = WhisperVadFactory.FromPath(args[3], new WhisperFactoryOptions { UseGpu = useGpu });
    Console.WriteLine($"LoadedLibrary={RuntimeOptions.LoadedLibrary}");
    var clip = File.ReadAllBytes(args[4]);
    var samples = new float[(clip.Length - 44) / 2];
    for (var i = 0; i < samples.Length; i++) samples[i] = BitConverter.ToInt16(clip, 44 + 2 * i) / 32768f;
    using var proc = vad.CreateBuilder().Build();
    var segs = proc.DetectSpeech(samples);
    Console.WriteLine($"VAD segments={segs.Count} " + string.Join(" ", segs.Select(s => $"[{s.Start.TotalSeconds:F2}-{s.End.TotalSeconds:F2}]")));
}
catch (Exception e) { Console.WriteLine("FAIL " + e.GetType().Name + ": " + e.Message.Split('\n')[0]); }

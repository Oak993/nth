static object Member(object obj, string name) => obj.GetType().GetProperty(name, BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Instance)!.GetValue(obj)!;
static void Select(object obj, string id)
{
    var method = obj.GetType().GetMethod("Select", BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Instance)!;
    var parameters = method.GetParameters().Select((p, i) => i == 0 ? (object)id : p.DefaultValue).ToArray();
    method.Invoke(obj, parameters);
}
static string F(double value) => value.ToString("G17", CultureInfo.InvariantCulture);
static void Emit(string stage, string path, string transition, int index, long ticks, double value, double previous) => Console.WriteLine(string.Join(",", stage, path, transition, index, F(ticks / 10000.0), F(value), F(previous - value), F((previous - value) * 100)));

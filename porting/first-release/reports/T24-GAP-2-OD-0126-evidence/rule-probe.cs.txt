using System.Globalization;
using System.Reflection;
using System.Text.Json;
using Marrow.App.Tests.Fixtures;
var assembly = typeof(HeadlessSession).Assembly;
var type = assembly.GetType("Marrow.App.Tests.Fixtures.FrameCurves")!;
var flags = BindingFlags.Static | BindingFlags.NonPublic;
var curve = type.GetField("Ease", flags)!.GetValue(null)!;
var frameMs = (double)type.GetField("FrameMs", flags)!.GetRawConstantValue()!;
var method = type.GetMethod("Bezier", flags)!;
double Reference(double x) => (double)method.Invoke(null, [curve, Math.Min(1,x)])!;
var output = new List<object>();
foreach (var line in File.ReadLines(args[0]).Skip(1))
{
    var columns=line.Split(',');
    if (columns[1]!="page" || int.Parse(columns[3],CultureInfo.InvariantCulture)==0) continue;
    int frame=int.Parse(columns[3],CultureInfo.InvariantCulture);
    double step=double.Parse(columns[6],CultureInfo.InvariantCulture);
    var transition=columns[2];
    double designStep=transition=="detail-out" ? Reference(frame*frameMs/1000/.1)-Reference((frame-1)*frameMs/1000/.1) : 0;
    bool allowed=step<=.15+1e-9 || step<=designStep;
    output.Add(new { transition,frame,actual_elapsed_ms=double.Parse(columns[4],CultureInfo.InvariantCulture),step,designStep,curve_extra_margin=0,allowed });
}
Console.WriteLine(JsonSerializer.Serialize(output,new JsonSerializerOptions{WriteIndented=true}));
Environment.ExitCode=output.Any(x=>(bool)x.GetType().GetProperty("allowed")!.GetValue(x)! == false)?1:0;
